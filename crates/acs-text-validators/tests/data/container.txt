thumbnails
{
	icon thumbnails
	menu-token "$ccp_thumbnails_menu-name"
	description "" 
	unique 1
	default-amount 1
	kind "structure"
  array-element
  {
    kind-value-A "thumbnails-element"
    kind-value-B "conditions"
  }
  validation
  {
    UniqueNames
  } 
}

thumbnails-element
{
  icon thumbnails-element
  menu-token "$ccp_thumbnails-element_menu-name"
  description ""
  kind "array"
  subpossibilities
  {
    image
    {
      kind value
      type filepathedit
      datatype "image"
      compulsory 3.4
    }
    width
    {
      kind value
      type int
      compulsory 3.4
    }
    height
    {
      kind value
      type int
      compulsory 3.4
    }
    example
    {
      kind element
      element-type "conditions"
    }
  }
}
